cmake_minimum_required(VERSION 3.18)
project(mea_reader LANGUAGES CXX)
option(WITH_MAXLAB "Build the Linux MaxLab SDK adapter" OFF)
add_executable(mea_reader reader.cpp)
target_compile_features(mea_reader PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(mea_reader PRIVATE /W4)
else()
  target_compile_options(mea_reader PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WITH_MAXLAB)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The supplied MaxLab static library requires Linux")
  endif()
  set(MAXLAB_ROOT "" CACHE PATH "SDK maxlab directory containing include/ and lib/")
  find_path(MAXLAB_INCLUDE maxlab/maxlab.h PATHS "${MAXLAB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(MAXLAB_LIBRARY maxlab PATHS "${MAXLAB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  find_package(Threads REQUIRED)
  target_include_directories(mea_reader PRIVATE "${MAXLAB_INCLUDE}")
  target_link_libraries(mea_reader PRIVATE "${MAXLAB_LIBRARY}" Threads::Threads)
  target_compile_definitions(mea_reader PRIVATE WITH_MAXLAB=1)
endif()
